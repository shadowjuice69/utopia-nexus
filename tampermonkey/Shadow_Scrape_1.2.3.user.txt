// ==UserScript==
// @name         Shadow Scrape
// @namespace    utopia-nexus-shadow-scrape
// @version      1.2.3
// @description  Automatic Utopia page capture for Nexus, with optional kingdom-details auto-crawl (island:kingdom walk)
// @match        https://www.utopia-game.com/*
// @match        https://utopia-game.com/*
// @match        https://intel.utopia-game.com/*
// @match        https://intel.utopia.site/*
// @grant        GM_xmlhttpRequest
// @grant        GM_getValue
// @grant        GM_setValue
// @connect      utopia-nexus.onrender.com
// @run-at       document-end
// @all-frames   false
// ==/UserScript==

(function () {
    "use strict";

    const VERSION = "1.2.3";
    const SOURCE = "shadow-scrape";
    const ENDPOINT = "https://utopia-nexus.onrender.com/intel";

    const KEY_NAME = "shadow_scrape_nexus_key";
    const CRAWL_RUNNING_KEY = "shadow_scrape_crawl_running";
    const CRAWL_TOTAL_KEY = "shadow_scrape_crawl_total";
    const CRAWL_LAST_KEY = "shadow_scrape_crawl_last_processed";

    const CAPTURE_DELAY = 2500;
    const RETRIES = 3;
    const RETRY_DELAY = 3000;
    const CRAWL_STEP_DELAY = 10000;

    // Kingdoms run 1 through 12.
    // 1:12 -> 2:1, 2:12 -> 3:1, etc.
    const MAX_KINGDOM = 12;

    let lastUrl = "";
    let captureTimer = null;
    let navigationTimer = null;

    function getKey() {
        return GM_getValue(KEY_NAME, "");
    }

    function isCrawlRunning() {
        return GM_getValue(CRAWL_RUNNING_KEY, false) === true;
    }

    function setCrawlRunning(value) {
        GM_setValue(CRAWL_RUNNING_KEY, !!value);
    }

    function getCrawlTotal() {
        return Number(GM_getValue(CRAWL_TOTAL_KEY, 0)) || 0;
    }

    function setCrawlTotal(value) {
        GM_setValue(CRAWL_TOTAL_KEY, Number(value) || 0);
    }

    function getCrawlLastProcessed() {
        return GM_getValue(CRAWL_LAST_KEY, "");
    }

    function setCrawlLastProcessed(value) {
        GM_setValue(CRAWL_LAST_KEY, String(value || ""));
    }

    function bumpCrawlTotal() {
        const next = getCrawlTotal() + 1;
        setCrawlTotal(next);
        return next;
    }

    function sleep(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }

    function cleanText(value) {
        return String(value || "").replace(/\s+/g, " ").trim();
    }

    function nowIso() {
        return new Date().toISOString();
    }

    function makeCaptureId() {
        return [Date.now(), Math.random().toString(36).slice(2, 10)].join("-");
    }

    function buildSafeDom() {
        try {
            const clone = document.documentElement.cloneNode(true);
            const removeSelectors = ["script", "style", "noscript", "iframe", "object", "embed", "canvas", "video", "audio"];

            for (const selector of removeSelectors) {
                clone.querySelectorAll(selector).forEach(node => node.remove());
            }

            clone.querySelectorAll("input, textarea, select").forEach(node => {
                try {
                    node.removeAttribute("value");
                    node.removeAttribute("name");
                } catch (_) {}
            });

            clone.querySelectorAll("*").forEach(node => {
                try {
                    node.removeAttribute("onclick");
                    node.removeAttribute("onchange");
                    node.removeAttribute("onsubmit");
                    node.removeAttribute("onload");
                    node.removeAttribute("onerror");
                    node.removeAttribute("src");
                    node.removeAttribute("srcset");
                    if (node.tagName === "A") node.removeAttribute("href");
                } catch (_) {}
            });

            return clone.outerHTML;
        } catch (error) {
            console.warn("[Shadow Scrape] safe DOM failed", error);
            return document.documentElement ? document.documentElement.outerHTML : "";
        }
    }

    function identityHints() {
        const bodyText = cleanText(document.body ? document.body.innerText : "");
        let province = "";
        let kd = "";

        const provincePatterns = [
            /(?:province|prov)\s*[:#-]?\s*([A-Za-z0-9 _'-]{2,80})/i,
            /(?:you are|logged in as)\s+([A-Za-z0-9 _'-]{2,80})/i
        ];

        for (const pattern of provincePatterns) {
            const match = bodyText.match(pattern);
            if (match && match[1]) {
                province = cleanText(match[1]);
                break;
            }
        }

        const kdMatch = bodyText.match(/\b(\d{1,3})\s*:\s*(\d{1,3})\b/);
        if (kdMatch) kd = `${kdMatch[1]}:${kdMatch[2]}`;

        return { province, kd };
    }

    function parseUrlKd() {
        const match = String(location.href || "").match(/\/kingdom_details\/(\d+)\/(\d+)(?:[\/?#]|$)/i);
        if (!match) return null;
        return { island: Number(match[1]), kingdom: Number(match[2]) };
    }

    function buildKdUrl(island, kingdom) {
        return `${location.origin}/wol/game/kingdom_details/${island}/${kingdom}`;
    }

    function navigateToKd(island, kingdom) {
        const url = buildKdUrl(island, kingdom);
        console.log(`[Shadow Scrape] Crawl navigation → ${island}:${kingdom}`);
        window.location.href = url;
    }

    function buildCapture() {
        const identity = identityHints();
        let visibleText = "";

        try {
            visibleText = cleanText(document.body ? document.body.innerText : "");
        } catch (_) {}

        const kd = parseUrlKd();

        return {
            title: document.title || "",
            url: location.href,
            hostname: location.hostname,
            pathname: location.pathname,
            captured_at: nowIso(),
            identity: { province: identity.province, kd: identity.kd },
            kingdom_details: kd ? { island: kd.island, kingdom: kd.kingdom } : null,
            visible_text: visibleText,
            raw_dom: buildSafeDom()
        };
    }

    function encode(record) {
        return {
            key: getKey(),
            source: SOURCE,
            version: VERSION,
            tab: document.title || "",
            prov: record.identity ? record.identity.province : "",
            kd: record.identity ? record.identity.kd : "",
            url: record.url,
            capture_id: makeCaptureId(),
            captured_at: record.captured_at,
            data_simple: record
        };
    }

    function sendOnce(payload) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: "POST",
                url: ENDPOINT,
                headers: { "Content-Type": "application/json" },
                data: JSON.stringify(payload),
                timeout: 30000,
                onload(response) {
                    if (response.status >= 200 && response.status < 300) {
                        resolve(response);
                        return;
                    }
                    reject(new Error(`HTTP ${response.status}: ${response.responseText || ""}`));
                },
                onerror() { reject(new Error("Network error")); },
                ontimeout() { reject(new Error("Request timed out")); }
            });
        });
    }

    async function sendWithRetry(payload) {
        let lastError = null;
        for (let attempt = 1; attempt <= RETRIES; attempt++) {
            try {
                return await sendOnce(payload);
            } catch (error) {
                lastError = error;
                console.warn(`[Shadow Scrape] send attempt ${attempt}/${RETRIES} failed`, error);
                if (attempt < RETRIES) await sleep(RETRY_DELAY);
            }
        }
        throw lastError || new Error("Send failed");
    }

    function handleCrawlAdvance() {
        if (!isCrawlRunning()) return;

        const here = parseUrlKd();
        if (!here) return;

        const currentKey = `${here.island}:${here.kingdom}`;
        const total = bumpCrawlTotal();
        setCrawlLastProcessed(currentKey);

        let nextIsland = here.island;
        let nextKingdom = here.kingdom + 1;

        // Never request kingdom 13.
        // 1:12 -> 2:1, 2:12 -> 3:1, 3:12 -> 4:1.
        if (here.kingdom >= MAX_KINGDOM) {
            nextIsland = here.island + 1;
            nextKingdom = 1;
        }

        console.log(`[Shadow Scrape] Crawl complete: ${currentKey}`);
        console.log(`[Shadow Scrape] Crawl total: ${total}`);
        console.log(`[Shadow Scrape] Next kingdom: ${nextIsland}:${nextKingdom}`);

        setTimeout(() => {
            if (isCrawlRunning()) navigateToKd(nextIsland, nextKingdom);
        }, CRAWL_STEP_DELAY);
    }

    function startCrawl() {
        const current = parseUrlKd();
        if (!current) {
            alert("Open a kingdom details page first.\n\nExample:\nKingdom 1:1");
            return;
        }

        setCrawlTotal(0);
        setCrawlLastProcessed("");
        setCrawlRunning(true);

        console.log(`[Shadow Scrape] Kingdom crawler STARTED at ${current.island}:${current.kingdom}`);
        scheduleCapture(500);
    }

    function stopCrawl() {
        setCrawlRunning(false);
        console.log("[Shadow Scrape] Kingdom crawler STOPPED");
    }

    function resetCrawl() {
        setCrawlRunning(false);
        setCrawlTotal(0);
        setCrawlLastProcessed("");
        console.log("[Shadow Scrape] Kingdom crawler RESET");
    }

    function renderCrawlToggle() {
        if (document.getElementById("shadow-scrape-crawl-panel")) return;

        const panel = document.createElement("div");
        panel.id = "shadow-scrape-crawl-panel";
        panel.style.cssText = "position:fixed;right:12px;bottom:12px;z-index:2147483647;background:rgba(20,20,20,.95);color:#fff;padding:10px;border:1px solid rgba(255,255,255,.2);border-radius:8px;font-family:Arial,sans-serif;font-size:12px;box-shadow:0 4px 18px rgba(0,0,0,.4);";

        const title = document.createElement("div");
        title.textContent = `Shadow Scrape ${VERSION}`;
        title.style.cssText = "font-weight:bold;margin-bottom:6px;";
        panel.appendChild(title);

        const status = document.createElement("div");
        status.id = "shadow-scrape-crawl-status";
        status.style.marginBottom = "8px";
        panel.appendChild(status);

        const startButton = document.createElement("button");
        startButton.textContent = "Start Cycler";
        startButton.style.marginRight = "5px";
        startButton.onclick = () => { startCrawl(); updateCrawlStatus(); };
        panel.appendChild(startButton);

        const stopButton = document.createElement("button");
        stopButton.textContent = "Stop";
        stopButton.style.marginRight = "5px";
        stopButton.onclick = () => { stopCrawl(); updateCrawlStatus(); };
        panel.appendChild(stopButton);

        const resetButton = document.createElement("button");
        resetButton.textContent = "Reset";
        resetButton.onclick = () => { resetCrawl(); updateCrawlStatus(); };
        panel.appendChild(resetButton);

        document.body.appendChild(panel);
        updateCrawlStatus();
    }

    function updateCrawlStatus() {
        const status = document.getElementById("shadow-scrape-crawl-status");
        if (!status) return;

        const running = isCrawlRunning();
        const total = getCrawlTotal();
        const last = getCrawlLastProcessed();
        const current = parseUrlKd();

        status.textContent = running
            ? `Cycler: ON | Captured: ${total} | Last: ${last || "none"} | Current: ${current ? `${current.island}:${current.kingdom}` : "not KD"}`
            : `Cycler: OFF | Captured: ${total} | Last: ${last || "none"}`;
    }

    async function capture() {
        const currentUrl = location.href;
        if (lastUrl === currentUrl) return;

        lastUrl = currentUrl;
        console.log("[Shadow Scrape] Capturing:", currentUrl);

        const record = buildCapture();
        const payload = encode(record);

        try {
            await sendWithRetry(payload);
            console.log("[Shadow Scrape] Saved:", currentUrl);
            updateCrawlStatus();

            if (isCrawlRunning() && parseUrlKd()) {
                handleCrawlAdvance();
            }
        } catch (error) {
            console.error("[Shadow Scrape] Capture failed:", error);
        }
    }

    function scheduleCapture(delay = CAPTURE_DELAY) {
        if (captureTimer) clearTimeout(captureTimer);
        captureTimer = setTimeout(() => capture(), delay);
    }

    function hookNavigation() {
        const originalPushState = history.pushState;
        history.pushState = function () {
            const result = originalPushState.apply(this, arguments);
            navigationChanged();
            return result;
        };

        const originalReplaceState = history.replaceState;
        history.replaceState = function () {
            const result = originalReplaceState.apply(this, arguments);
            navigationChanged();
            return result;
        };

        window.addEventListener("popstate", navigationChanged);
        window.addEventListener("hashchange", navigationChanged);
    }

    function navigationChanged() {
        if (navigationTimer) clearTimeout(navigationTimer);
        lastUrl = "";

        navigationTimer = setTimeout(() => {
            renderCrawlToggle();
            scheduleCapture(800);
        }, 800);
    }

    function observePage() {
        const observer = new MutationObserver(() => {
            if (!document.getElementById("shadow-scrape-crawl-panel")) {
                renderCrawlToggle();
            }
        });

        if (document.documentElement) {
            observer.observe(document.documentElement, {
                childList: true,
                subtree: true
            });
        }
    }

    function start() {
        hookNavigation();
        observePage();
        scheduleCapture();
        renderCrawlToggle();

        console.log("[Shadow Scrape] active", location.hostname, `v${VERSION}`);

        if (isCrawlRunning() && parseUrlKd()) {
            setTimeout(() => {
                lastUrl = "";
                capture();
            }, 800);
        }
    }

    start();

})();
