// ==UserScript==
// @name         Nexus KD Crawler
// @namespace    utopia-nexus-kd-crawler
// @version      1.0.1
// @description  Auto-walks kingdom_details pages (island:kingdom), captures roster intel, advances automatically, stops when it runs off the edge of the game world
// @match        https://www.utopia-game.com/*
// @match        https://utopia-game.com/*
// @grant        GM_xmlhttpRequest
// @grant        GM_getValue
// @grant        GM_setValue
// @connect      utopia-nexus.onrender.com
// @run-at       document-end
// @all-frames   false
// ==/UserScript==

(function () {
  "use strict";

  const VERSION = "1.0.1";
  const ENDPOINT = "https://utopia-nexus.onrender.com/intel";
  const KEY = "Nikko" + "Ace";

  const STEP_DELAY_MS = 10000; // 10s between pages, per your "slow/safe" choice
  const PAGE_READY_TIMEOUT_MS = 8000;
  const PAGE_READY_POLL_MS = 300;
  const MAX_KINGDOM = 12;

  const ST_RUNNING = "kdcrawler_running";
  const ST_TOTAL_CAPTURED = "kdcrawler_total_captured";
  const ST_LAST_PROCESSED_URL = "kdcrawler_last_processed_url";
  const ST_LOG = "kdcrawler_log"; // small rolling log for the status panel

  function isRunning() {
    return GM_getValue(ST_RUNNING, false) === true;
  }

  function setRunning(val) {
    GM_setValue(ST_RUNNING, !!val);
  }

  function bumpCaptured() {
    const n = GM_getValue(ST_TOTAL_CAPTURED, 0) + 1;
    GM_setValue(ST_TOTAL_CAPTURED, n);
    return n;
  }

  function addLog(line) {
    const log = GM_getValue(ST_LOG, []);
    log.unshift(`${new Date().toLocaleTimeString()} â€” ${line}`);
    GM_setValue(ST_LOG, log.slice(0, 8));
  }

  function parseUrlKd() {
    const m = location.pathname.match(/kingdom_details\/(\d+)\/(\d+)/);
    if (!m) return null;
    return { island: parseInt(m[1], 10), kingdom: parseInt(m[2], 10) };
  }

  function buildKdUrl(island, kingdom) {
    return `${location.origin}/wol/game/kingdom_details/${island}/${kingdom}`;
  }

  function getNextKd(island, kingdom) {
    if (kingdom >= MAX_KINGDOM) return { island: island + 1, kingdom: 1 };
    return { island, kingdom: kingdom + 1 };
  }

  function getBodyText() {
    return document.body ? document.body.innerText || "" : "";
  }

  function extractTotalProvinces(text) {
    const m = text.match(/Total\s+Provinces\s*[\r\n]*\s*([\d,]+)/i);
    if (!m) return null;
    return parseInt(m[1].replace(/,/g, ""), 10);
  }

  function extractKdName(text) {
    const m = text.match(/kingdom of\s+(.+?)\s*\(\d+:\d+\)/i);
    return m ? m[1].trim() : "Unknown";
  }

  function extractKdCodeFromHeading(text) {
    const m = text.match(/kingdom of\s+.+?\((\d+:\d+)\)/i);
    return m ? m[1] : "";
  }

  function extractRoster() {
    const table =
      document.getElementById("clone_kingdom") ||
      document.querySelector("table.tablesorter");
    if (!table) return [];

    const headerCells = Array.from(
      table.querySelectorAll("thead th, tr:first-child th, tr:first-child td")
    ).map((c) => c.textContent.trim().toLowerCase());

    const colIndex = (names) => {
      for (const name of names) {
        const idx = headerCells.findIndex((h) => h.includes(name));
        if (idx !== -1) return idx;
      }
      return -1;
    };

    const idx = {
      province: colIndex(["province"]),
      race: colIndex(["race"]),
      land: colIndex(["land"]),
      networth: colIndex(["net worth", "networth"]),
      nwpa: colIndex(["nwpa"]),
      honor: colIndex(["nobility", "honor"]),
      gains: colIndex(["gains"]),
    };

    const rows = Array.from(table.querySelectorAll("tbody tr, tr")).filter(
      (tr) => tr.querySelectorAll("td").length >= 3
    );

    const roster = [];
    for (const row of rows) {
      const cells = Array.from(row.querySelectorAll("td")).map((td) =>
        td.textContent.trim()
      );
      if (!cells.length) continue;
      const province = idx.province !== -1 ? cells[idx.province] : "";
      if (!province) continue;
      roster.push({
        province,
        race: idx.race !== -1 ? cells[idx.race] : "",
        land: idx.land !== -1 ? cells[idx.land] : "",
        networth: idx.networth !== -1 ? cells[idx.networth] : "",
        nwpa: idx.nwpa !== -1 ? cells[idx.nwpa] : "",
        honor: idx.honor !== -1 ? cells[idx.honor] : "",
        gains: idx.gains !== -1 ? cells[idx.gains] : "",
      });
    }
    return roster;
  }

  function renderPanel() {
    let panel = document.getElementById("kd-crawler-panel");
    if (!panel) {
      panel = document.createElement("div");
      panel.id = "kd-crawler-panel";
      panel.style.cssText = [
        "position:fixed",
        "bottom:16px",
        "left:16px",
        "z-index:2147483647",
        "background:#0d1117",
        "color:#e6edf3",
        "border:1px solid #30363d",
        "border-radius:10px",
        "padding:10px 14px",
        "font:12px/1.5 monospace",
        "max-width:360px",
        "box-shadow:0 4px 18px rgba(0,0,0,.5)",
      ].join(";");
      document.documentElement.appendChild(panel);
    }

    const running = isRunning();
    const total = GM_getValue(ST_TOTAL_CAPTURED, 0);
    const log = GM_getValue(ST_LOG, []);
    const here = parseUrlKd();
    const hereLabel = here ? `${here.island}:${here.kingdom}` : "(not on a kd page)";

    panel.innerHTML = `
      <div style="font-weight:700;margin-bottom:4px;">âš¡ Nexus KD Crawler v${VERSION}</div>
      <div>Status: <b style="color:${running ? "#3fb950" : "#8b949e"}">${
      running ? "RUNNING" : "stopped"
    }</b></div>
      <div>Current page: ${hereLabel}</div>
      <div>Captured this run: ${total}</div>
      <div style="margin-top:6px;">
        <button id="kd-crawler-toggle" style="cursor:pointer;padding:4px 10px;border-radius:6px;border:1px solid #30363d;background:${
          running ? "#da3633" : "#238636"
        };color:white;font:inherit;">${running ? "Stop" : "Start from here"}</button>
        <button id="kd-crawler-reset" style="cursor:pointer;padding:4px 10px;border-radius:6px;border:1px solid #30363d;background:#21262d;color:white;font:inherit;margin-left:6px;">Reset count</button>
      </div>
      <div style="margin-top:6px;max-height:110px;overflow:auto;color:#8b949e;">
        ${log.map((l) => `<div>${l}</div>`).join("")}
      </div>
    `;

    document.getElementById("kd-crawler-toggle").onclick = () => {
      if (isRunning()) {
        setRunning(false);
        addLog("Stopped by user");
        renderPanel();
      } else {
        setRunning(true);
        addLog(`Started at ${hereLabel}`);
        renderPanel();
        setTimeout(processCurrentPage, 500);
      }
    };

    document.getElementById("kd-crawler-reset").onclick = () => {
      GM_setValue(ST_TOTAL_CAPTURED, 0);
      addLog("Counter reset");
      renderPanel();
    };
  }

  function sendKingdomPage(payload) {
    return new Promise((resolve) => {
      const form = new URLSearchParams();
      form.set("key", KEY);
      form.set("source", "kd-crawler");
      form.set("tab", "kingdom_details");
      form.set("kd", payload.kd);
      form.set("prov", "");
      form.set("data_simple", JSON.stringify(payload));

      GM_xmlhttpRequest({
        method: "POST",
        url: ENDPOINT,
        headers: { "Content-Type": "application/x-www-form-urlencoded" },
        data: form.toString(),
        timeout: 30000,
        onload: (r) => resolve({ ok: r.status >= 200 && r.status < 300, status: r.status }),
        onerror: () => resolve({ ok: false, status: 0 }),
        ontimeout: () => resolve({ ok: false, status: 0 }),
      });
    });
  }

  function waitForReady() {
    return new Promise((resolve, reject) => {
      const start = Date.now();
      const check = () => {
        const text = getBodyText();
        if (/Total\s+Provinces/i.test(text)) {
          resolve(text);
          return;
        }
        if (Date.now() - start > PAGE_READY_TIMEOUT_MS) {
          reject(new Error("Timed out waiting for kingdom stats to render"));
          return;
        }
        setTimeout(check, PAGE_READY_POLL_MS);
      };
      check();
    });
  }

  async function processCurrentPage() {
    if (!isRunning()) return;

    const here = parseUrlKd();
    if (!here) {
      addLog("Not on a kingdom_details page â€” stopped");
      setRunning(false);
      renderPanel();
      return;
    }

    const urlKey = `${here.island}:${here.kingdom}`;
    if (GM_getValue(ST_LAST_PROCESSED_URL, "") === urlKey) {
      return;
    }

    let text;
    try {
      text = await waitForReady();
    } catch (e) {
      addLog(`âœ• ${urlKey}: ${e.message} â€” stopped`);
      setRunning(false);
      renderPanel();
      return;
    }

    const provinces = extractTotalProvinces(text);

    if (provinces === null) {
      addLog(`âœ• ${urlKey}: couldn't read Total Provinces â€” stopped`);
      setRunning(false);
      renderPanel();
      return;
    }

    if (provinces === 0) {
      if (here.kingdom === 1) {
        addLog(`â—‹ ${urlKey}: 0 provs on a fresh island â€” no more kingdoms found. Crawl complete.`);
        GM_setValue(ST_LAST_PROCESSED_URL, urlKey);
        setRunning(false);
        renderPanel();
        return;
      }
      addLog(`â—‹ ${urlKey}: 0 provs â€” island exhausted, moving to ${here.island + 1}:1`);
      GM_setValue(ST_LAST_PROCESSED_URL, urlKey);
      renderPanel();
      setTimeout(() => {
        if (isRunning()) location.href = buildKdUrl(here.island + 1, 1);
      }, STEP_DELAY_MS);
      return;
    }

    const kdCode =
      extractKdCodeFromHeading(text) || urlKey;
    const payload = {
      kd: kdCode,
      kd_name: extractKdName(text),
      total_provinces: provinces,
      url: location.href,
      captured_at: new Date().toISOString(),
      provinces: extractRoster(),
    };

    const result = await sendKingdomPage(payload);
    GM_setValue(ST_LAST_PROCESSED_URL, urlKey);

    if (result.ok) {
      const total = bumpCaptured();
      addLog(`âœ“ ${urlKey} "${payload.kd_name}" â€” ${payload.provinces.length} rows saved (total: ${total})`);
    } else {
      addLog(`âœ• ${urlKey}: send failed (status ${result.status}) â€” continuing anyway`);
    }

    renderPanel();

    const next = getNextKd(here.island, here.kingdom);
    setTimeout(() => {
      if (isRunning()) location.href = buildKdUrl(next.island, next.kingdom);
    }, STEP_DELAY_MS);
  }

  function start() {
    renderPanel();
    if (isRunning() && parseUrlKd()) {
      setTimeout(processCurrentPage, 800);
    }
    console.log("[Nexus KD Crawler] loaded", location.pathname);
  }

  start();
})();
