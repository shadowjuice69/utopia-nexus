// ==UserScript==
// @name         Shadow Scrape
// @namespace    utopia-nexus-shadow-scrape
// @version      1.1.1
// @description  Automatic Utopia page capture for Nexus + Kingdom Cycler
// @match        https://www.utopia-game.com/*
// @match        https://utopia-game.com/*
// @match        https://intel.utopia-game.com/*
// @match        https://intel.utopia.site/*
// @grant        GM_xmlhttpRequest
// @grant        GM_getValue
// @grant        GM_setValue
// @connect      utopia-nexus.onrender.com
// @run-at       document-end
// @all-frames   false
// ==/UserScript==

(function () {
    "use strict";

    const VERSION = "1.1.1";
    const SOURCE = "shadow-scrape";
    const ENDPOINT = "https://utopia-nexus.onrender.com/intel";

    const KEY_STORAGE = "shadow_scrape_nexus_key";
    const CAPTURE_DELAY = 2500;
    const RETRIES = 3;
    const RETRY_DELAY = 3000;
    const KINGDOM_CYCLER_KEY = "shadow_scrape_kingdom_cycler";

    let busy = false;
    let lastUrl = "";
    let kingdomAdvanceScheduled = false;

    function toast(message, success = true, duration = 4500) {
        const old = document.getElementById("shadow-scrape-status");
        if (old) old.remove();
        const el = document.createElement("div");
        el.id = "shadow-scrape-status";
        el.textContent = message;
        el.style.cssText = [
            "position:fixed","top:16px","right:16px","z-index:2147483647",
            "padding:10px 15px","border-radius:8px","font:600 12px monospace",
            "color:white",`background:${success ? "#238636" : "#da3633"}`,
            "box-shadow:0 4px 18px rgba(0,0,0,.45)","max-width:420px",
            "pointer-events:none","white-space:pre-wrap"
        ].join(";");
        document.documentElement.appendChild(el);
        setTimeout(() => { if (el.parentNode) el.remove(); }, duration);
    }

    function getNexusKey() {
        let key = GM_getValue(KEY_STORAGE, "");
        if (key && typeof key === "string") return key.trim();
        key = window.prompt(
            "Shadow Scrape\n\nEnter your Nexus Intel key.\n\nIt will be stored locally in Tampermonkey and will not be sent anywhere except the Nexus /intel endpoint."
        );
        if (!key) return "";
        key = key.trim();
        if (!key) return "";
        GM_setValue(KEY_STORAGE, key);
        return key;
    }

    function captureId() {
        if (typeof crypto !== "undefined" && crypto.randomUUID) return crypto.randomUUID();
        return "shadow-" + Date.now() + "-" + Math.random().toString(36).slice(2);
    }

    function cleanText(text) {
        return String(text || "")
            .replace(/\u00a0/g, " ")
            .replace(/\r\n?/g, "\n")
            .replace(/[ \t]+\n/g, "\n")
            .replace(/\n{3,}/g, "\n\n")
            .trim();
    }

    function buildSafeDom() {
        const source = document.documentElement;
        if (!source) return "";
        const clone = source.cloneNode(true);
        clone.querySelectorAll("script,style,noscript,template,iframe").forEach(el => el.remove());
        clone.querySelectorAll("input,textarea,select").forEach(el => {
            el.removeAttribute("value");
            el.removeAttribute("data-value");
            if (el.tagName === "TEXTAREA") el.textContent = "";
            if (el.tagName === "SELECT") el.querySelectorAll("option").forEach(option => option.removeAttribute("selected"));
        });
        clone.querySelectorAll("*").forEach(el => {
            ["onclick","onchange","oninput","onsubmit","onload","onerror","onkeydown","onkeyup","onkeypress"].forEach(attr => el.removeAttribute(attr));
            Array.from(el.attributes).forEach(attr => {
                const name = attr.name.toLowerCase();
                if (name.includes("token") || name.includes("password") || name.includes("passwd") || name.includes("secret") || name.includes("credential") || name.includes("session") || name.includes("auth")) el.removeAttribute(attr.name);
            });
        });
        return clone.outerHTML;
    }

    function identityHints(text) {
        const identity = {};
        const province = text.match(/(?:Province:\s*|The Province of\s+)([^\n(]{2,80})\s*\(?([0-9]+:[0-9]+)?/i);
        if (province) {
            identity.province = province[1].trim();
            if (province[2]) identity.kd_code = province[2];
        }
        if (!identity.kd_code) {
            const kd = text.match(/\b(\d+):(\d+)\b/);
            if (kd) identity.kd_code = `${kd[1]}:${kd[2]}`;
        }
        return identity;
    }

    function buildCapture() {
        const visibleText = cleanText(document.body?.innerText || "");
        const raw = buildSafeDom();
        const capturedAt = new Date().toISOString();
        return {
            capture_id: captureId(),
            captured_at: capturedAt,
            scraper_version: VERSION,
            page: {
                url: location.href, origin: location.origin, hostname: location.hostname,
                pathname: location.pathname, search: location.search, hash: location.hash,
                title: document.title || "", language: document.documentElement?.lang || null,
                ready_state: document.readyState
            },
            source_identity: { hostname: location.hostname, origin: location.origin },
            subject_identity: identityHints(visibleText),
            visible_text: visibleText,
            raw
        };
    }

    function encode(record, key) {
        const data = {
            category: SOURCE, rows: [], capture_id: record.capture_id,
            captured_at: record.captured_at, scraper_version: record.scraper_version,
            page: record.page, source_identity: record.source_identity,
            subject_identity: record.subject_identity, visible_text: record.visible_text,
            raw: record.raw
        };
        const form = new URLSearchParams();
        form.set("key", key); form.set("source", SOURCE); form.set("tab", "shadow");
        form.set("prov", record.subject_identity.province || "");
        form.set("kd", record.subject_identity.kd_code || "");
        form.set("url", record.page.url); form.set("capture_id", record.capture_id);
        form.set("captured_at", record.captured_at);
        form.set("data_simple", JSON.stringify(data));
        return form.toString();
    }

    function send(record, key) {
        return new Promise(resolve => {
            const body = encode(record, key);
            GM_xmlhttpRequest({
                method: "POST", url: ENDPOINT,
                headers: { "Content-Type": "application/x-www-form-urlencoded" },
                data: body, timeout: 30000,
                onload(response) {
                    let responseText = String(response.responseText || "").replace(/\s+/g, " ").trim();
                    if (responseText.length > 300) responseText = responseText.slice(0, 300) + "...";
                    resolve({ ok: response.status >= 200 && response.status < 300, status: response.status, statusText: response.statusText || "", responseText, bodyBytes: new Blob([body]).size });
                },
                onerror(error) { resolve({ ok: false, status: 0, statusText: "Network error", responseText: error?.error || error?.message || "", bodyBytes: new Blob([body]).size }); },
                ontimeout() { resolve({ ok: false, status: 0, statusText: "Timeout", responseText: "", bodyBytes: new Blob([body]).size }); }
            });
        });
    }

    async function sendWithRetry(record, key) {
        let lastResult = null;
        for (let attempt = 1; attempt <= RETRIES; attempt++) {
            toast(`Shadow Scrape → sending (${attempt}/${RETRIES})`);
            const result = await send(record, key);
            lastResult = result;
            if (result.ok) return result;
            console.warn("[Shadow Scrape] capture rejected", result);
            if (attempt < RETRIES) await new Promise(resolve => setTimeout(resolve, RETRY_DELAY));
        }
        return lastResult || { ok: false, status: 0, statusText: "Unknown error", responseText: "" };
    }

    function describeFailure(result) {
        if (!result) return "unknown failure";
        if (result.status === 401) return "401 Unauthorized — Nexus key rejected";
        if (result.status === 400) return "400 Bad Request — capture format rejected";
        if (result.status === 413) return "413 Payload Too Large — capture is too large";
        if (result.status === 500) return "500 Server Error — " + (result.responseText || "Render processing failed");
        if (result.status === 502 || result.status === 503 || result.status === 504) return `${result.status} — Render service unavailable`;
        if (result.status === 0) return "Network failure — " + (result.statusText || "request failed");
        return (`${result.status} ${result.statusText || ""}` + (result.responseText ? ` — ${result.responseText}` : "")).trim();
    }

    function getKingdomFromUrl() {
        const match = location.pathname.match(/\/wol\/game\/kingdom_details\/(\d+)\/(\d+)/i);
        if (!match) return null;
        return { world: Number(match[1]), kingdom: Number(match[2]), code: `${match[1]}:${match[2]}` };
    }

    function getKingdomCyclerState() {
        return GM_getValue(KINGDOM_CYCLER_KEY, { active: false });
    }

    function setKingdomCyclerActive(active) {
        GM_setValue(KINGDOM_CYCLER_KEY, { active: !!active });
    }

    function findNextKingdomHref() {
        const links = Array.from(document.querySelectorAll("a[href]"));
        const next = links.find(link => {
            const text = cleanText(link.innerText || link.textContent || "");
            const href = link.getAttribute("href") || "";
            return /^Next$/i.test(text) && /\/wol\/game\/kingdom_details\//i.test(href);
        });
        if (!next) return "";
        try { return new URL(next.getAttribute("href"), location.href).href; }
        catch (error) { console.warn("[Shadow Scrape] invalid kingdom Next link", error); return ""; }
    }

    function maybeStartKingdomCycler() {
        const kingdom = getKingdomFromUrl();
        if (!kingdom) return;
        if (kingdom.world === 1 && kingdom.kingdom === 1) {
            setKingdomCyclerActive(true);
            console.log("[Shadow Scrape] Kingdom Cycler started at", kingdom.code);
        }
    }

    function maybeAdvanceKingdom() {
        if (kingdomAdvanceScheduled) return;
        const kingdom = getKingdomFromUrl();
        if (!kingdom) return;
        const state = getKingdomCyclerState();
        if (!state.active) return;

        const nextHref = findNextKingdomHref();
        if (!nextHref) {
            setKingdomCyclerActive(false);
            toast(`Kingdom Cycler ✓ finished at ${kingdom.code}`, true, 8000);
            console.log("[Shadow Scrape] Kingdom Cycler finished", kingdom.code);
            return;
        }
        if (nextHref === location.href) {
            setKingdomCyclerActive(false);
            toast("Kingdom Cycler ✕ Next link points to this page", false, 8000);
            return;
        }

        kingdomAdvanceScheduled = true;
        toast(`Kingdom ${kingdom.code} ✓ saved\nNext → loading automatically`, true, 3500);
        console.log("[Shadow Scrape] Kingdom Cycler advancing", { current: kingdom.code, next: nextHref });
        setTimeout(() => { window.location.href = nextHref; }, 2500);
    }

    async function capture() {
        if (busy) return;
        if (lastUrl === location.href) return;
        busy = true;
        try {
            const key = getNexusKey();
            if (!key) { toast("Shadow Scrape ✕ Nexus key not configured", false, 6000); return; }
            const record = buildCapture();
            if (!record.raw) { toast("Shadow Scrape ✕ no DOM available", false); return; }
            toast("Shadow Scrape → captured");
            const result = await sendWithRetry(record, key);
            if (result.ok) {
                lastUrl = location.href;
                toast("Shadow Scrape ✓ Sent → Render received → saved");
                GM_setValue("shadow_scrape_last_capture", { url: location.href, capture_id: record.capture_id, captured_at: record.captured_at });
                console.log("[Shadow Scrape] SUCCESS", { url: location.href, capture_id: record.capture_id, status: result.status, bytes: result.bodyBytes, response: result.responseText });
                maybeAdvanceKingdom();
            } else {
                const failure = describeFailure(result);
                toast(`Shadow Scrape ✕ ${failure}`, false, 8000);
                console.error("[Shadow Scrape] FAILED", { url: location.href, status: result.status, statusText: result.statusText, response: result.responseText, payloadBytes: result.bodyBytes });
            }
        } catch (error) {
            console.error("[Shadow Scrape] exception", error);
            toast("Shadow Scrape ✕ capture exception", false, 7000);
        } finally { busy = false; }
    }

    function scheduleCapture() { setTimeout(capture, CAPTURE_DELAY); }

    function hookNavigation() {
        const originalPushState = history.pushState;
        const originalReplaceState = history.replaceState;
        history.pushState = function (...args) { const result = originalPushState.apply(this, args); setTimeout(capture, 1500); return result; };
        history.replaceState = function (...args) { const result = originalReplaceState.apply(this, args); setTimeout(capture, 1500); return result; };
        window.addEventListener("popstate", () => setTimeout(capture, 1500));
        window.addEventListener("hashchange", () => setTimeout(capture, 1500));
    }

    function observePage() {
        let timer = null;
        const observer = new MutationObserver(() => {
            clearTimeout(timer);
            timer = setTimeout(() => {
                if (document.body && location.href !== lastUrl) capture();
            }, 1200);
        });
        if (document.documentElement) observer.observe(document.documentElement, { childList: true, subtree: true });
    }

    function start() {
        hookNavigation();
        observePage();
        scheduleCapture();
        maybeStartKingdomCycler();
        console.log("[Shadow Scrape] active", location.hostname);
    }

    start();
})();
