// ==UserScript==
// @name         Shadow Scrape
// @namespace    utopia-nexus-shadow-scrape
// @version      1.1.3
// @description  Automatic Utopia page capture for Nexus + Kingdom Cycler
// @match        https://www.utopia-game.com/*
// @match        https://utopia-game.com/*
// @match        https://intel.utopia-game.com/*
// @match        https://intel.utopia.site/*
// @grant        GM_xmlhttpRequest
// @grant        GM_getValue
// @grant        GM_setValue
// @connect      utopia-nexus.onrender.com
// @run-at       document-end
// @all-frames   false
// ==/UserScript==

(function(){"use strict";
const VERSION="1.1.3",SOURCE="shadow-scrape",ENDPOINT="https://utopia-nexus.onrender.com/intel";
const KEY_STORAGE="shadow_scrape_nexus_key",CAPTURE_DELAY=2500,RETRIES=3,RETRY_DELAY=3000;
const KINGDOM_CYCLER_KEY="shadow_scrape_kingdom_cycler",KINGDOM_COUNT=93,KINGDOMS_PER_WORLD=13;
let busy=false,lastUrl="",kingdomAdvanceScheduled=false;
function toast(message,success=true,duration=4500){const old=document.getElementById("shadow-scrape-status");if(old)old.remove();const el=document.createElement("div");el.id="shadow-scrape-status";el.textContent=message;el.style.cssText=["position:fixed","top:16px","right:16px","z-index:2147483647","padding:10px 15px","border-radius:8px","font:600 12px monospace","color:white",`background:${success?"#238636":"#da3633"}`,"box-shadow:0 4px 18px rgba(0,0,0,.45)","max-width:420px","pointer-events:none","white-space:pre-wrap"].join(";");document.documentElement.appendChild(el);setTimeout(()=>{if(el.parentNode)el.remove()},duration)}
function getNexusKey(){let key=GM_getValue(KEY_STORAGE,"");if(key&&typeof key==="string")return key.trim();key=window.prompt("Shadow Scrape\n\nEnter your Nexus Intel key.\n\nIt will be stored locally in Tampermonkey and will not be sent anywhere except the Nexus /intel endpoint.");if(!key)return"";key=key.trim();if(!key)return"";GM_setValue(KEY_STORAGE,key);return key}
function captureId(){return typeof crypto!=="undefined"&&crypto.randomUUID?crypto.randomUUID():"shadow-"+Date.now()+"-"+Math.random().toString(36).slice(2)}
function cleanText(text){return String(text||"").replace(/\u00a0/g," ").replace(/\r\n?/g,"\n").replace(/[ \t]+\n/g,"\n").replace(/\n{3,}/g,"\n\n").trim()}
function buildSafeDom(){const source=document.documentElement;if(!source)return"";const clone=source.cloneNode(true);clone.querySelectorAll("script,style,noscript,template,iframe").forEach(el=>el.remove());clone.querySelectorAll("input,textarea,select").forEach(el=>{el.removeAttribute("value");el.removeAttribute("data-value");if(el.tagName==="TEXTAREA")el.textContent="";if(el.tagName==="SELECT")el.querySelectorAll("option").forEach(o=>o.removeAttribute("selected"))});clone.querySelectorAll("*").forEach(el=>{["onclick","onchange","oninput","onsubmit","onload","onerror","onkeydown","onkeyup","onkeypress"].forEach(a=>el.removeAttribute(a));Array.from(el.attributes).forEach(a=>{const n=a.name.toLowerCase();if(n.includes("token")||n.includes("password")||n.includes("passwd")||n.includes("secret")||n.includes("credential")||n.includes("session")||n.includes("auth"))el.removeAttribute(a.name)})});return clone.outerHTML}
function identityHints(text){const identity={};const province=text.match(/(?:Province:\s*|The Province of\s+)([^\n(]{2,80})\s*\(?([0-9]+:[0-9]+)?/i);if(province){identity.province=province[1].trim();if(province[2])identity.kd_code=province[2]}if(!identity.kd_code){const kd=text.match(/\b(\d+):(\d+)\b/);if(kd)identity.kd_code=`${kd[1]}:${kd[2]}`}return identity}
function buildCapture(){const visibleText=cleanText(document.body?.innerText||"");const raw=buildSafeDom();const capturedAt=new Date().toISOString();return{capture_id:captureId(),captured_at:capturedAt,scraper_version:VERSION,page:{url:location.href,origin:location.origin,hostname:location.hostname,pathname:location.pathname,search:location.search,hash:location.hash,title:document.title||"",language:document.documentElement?.lang||null,ready_state:document.readyState},source_identity:{hostname:location.hostname,origin:location.origin},subject_identity:identityHints(visibleText),visible_text:visibleText,raw}}
function encode(record,key){const data={category:SOURCE,rows:[],capture_id:record.capture_id,captured_at:record.captured_at,scraper_version:record.scraper_version,page:record.page,source_identity:record.source_identity,subject_identity:record.subject_identity,visible_text:record.visible_text,raw:record.raw};const form=new URLSearchParams();form.set("key",key);form.set("source",SOURCE);form.set("tab","shadow");form.set("prov",record.subject_identity.province||"");form.set("kd",record.subject_identity.kd_code||"");form.set("url",record.page.url);form.set("capture_id",record.capture_id);form.set("captured_at",record.captured_at);form.set("data_simple",JSON.stringify(data));return form.toString()}
function send(record,key){return new Promise(resolve=>{const body=encode(record,key);GM_xmlhttpRequest({method:"POST",url:ENDPOINT,headers:{"Content-Type":"application/x-www-form-urlencoded"},data:body,timeout:30000,onload(r){let t=String(r.responseText||"").replace(/\s+/g," ").trim();if(t.length>300)t=t.slice(0,300)+"...";resolve({ok:r.status>=200&&r.status<300,status:r.status,statusText:r.statusText||"",responseText:t,bodyBytes:new Blob([body]).size})},onerror(e){resolve({ok:false,status:0,statusText:"Network error",responseText:e?.error||e?.message||"",bodyBytes:new Blob([body]).size})},ontimeout(){resolve({ok:false,status:0,statusText:"Timeout",responseText:"",bodyBytes:new Blob([body]).size})}})})}
async function sendWithRetry(record,key){let lastResult=null;for(let attempt=1;attempt<=RETRIES;attempt++){toast(`Shadow Scrape → sending (${attempt}/${RETRIES})`);const result=await send(record,key);lastResult=result;if(result.ok)return result;console.warn("[Shadow Scrape] capture rejected",result);if(attempt<RETRIES)await new Promise(r=>setTimeout(r,RETRY_DELAY))}return lastResult||{ok:false,status:0,statusText:"Unknown error",responseText:""}}
function describeFailure(r){if(!r)return"unknown failure";if(r.status===401)return"401 Unauthorized — Nexus key rejected";if(r.status===400)return"400 Bad Request — capture format rejected";if(r.status===413)return"413 Payload Too Large — capture is too large";if(r.status===500)return"500 Server Error — "+(r.responseText||"Render processing failed");if([502,503,504].includes(r.status))return`${r.status} — Render service unavailable`;if(r.status===0)return"Network failure — "+(r.statusText||"request failed");return(`${r.status} ${r.statusText||""}`+(r.responseText?` — ${r.responseText}`:"" )).trim()}
function getKingdomFromUrl(){const m=location.pathname.match(/\/wol\/game\/kingdom_details\/(\d+)\/(\d+)/i);if(!m)return null;return{world:Number(m[1]),kingdom:Number(m[2]),code:`${m[1]}:${m[2]}`}}
function getKingdomCyclerState(){return GM_getValue(KINGDOM_CYCLER_KEY,{active:false})}
function setKingdomCyclerActive(active){GM_setValue(KINGDOM_CYCLER_KEY,{active:!!active})}
function maybeStartKingdomCycler(){const k=getKingdomFromUrl();if(!k)return;if(k.world===1&&k.kingdom===1){setKingdomCyclerActive(true);console.log("[Shadow Scrape] Kingdom Cycler started at",k.code);toast("Kingdom Cycler ✓ started at 1:1",true,3000)}}
function getNextKingdomUrl(k){if(!k)return"";const index=(k.world-1)*KINGDOMS_PER_WORLD+k.kingdom;if(index>=KINGDOM_COUNT)return"";const nextIndex=index+1;const world=Math.floor((nextIndex-1)/KINGDOMS_PER_WORLD)+1;const kingdom=((nextIndex-1)%KINGDOMS_PER_WORLD)+1;return new URL(`/wol/game/kingdom_details/${world}/${kingdom}`,location.origin).href}
function maybeAdvanceKingdom(){if(kingdomAdvanceScheduled)return;const k=getKingdomFromUrl();if(!k)return;if(!getKingdomCyclerState().active)return;const nextHref=getNextKingdomUrl(k);if(!nextHref){setKingdomCyclerActive(false);toast(`Kingdom Cycler ✓ finished at ${k.code}`,true,8000);console.log("[Shadow Scrape] Kingdom Cycler finished",k.code);return}kingdomAdvanceScheduled=true;const next=new URL(nextHref).pathname.match(/kingdom_details\/(\d+)\/(\d+)/i);const nextCode=next?`${next[1]}:${next[2]}`:"next";toast(`Kingdom ${k.code} ✓ saved\nNext → ${nextCode}`,true,3500);console.log("[Shadow Scrape] Kingdom Cycler advancing",{current:k.code,next:nextHref});setTimeout(()=>{window.location.href=nextHref},2500)}
async function capture(){if(busy||lastUrl===location.href)return;busy=true;try{const key=getNexusKey();if(!key){toast("Shadow Scrape ✕ Nexus key not configured",false,6000);return}const record=buildCapture();if(!record.raw){toast("Shadow Scrape ✕ no DOM available",false);return}toast("Shadow Scrape → captured");const result=await sendWithRetry(record,key);if(result.ok){lastUrl=location.href;toast("Shadow Scrape ✓ Sent → Render received → saved");GM_setValue("shadow_scrape_last_capture",{url:location.href,capture_id:record.capture_id,captured_at:record.captured_at});console.log("[Shadow Scrape] SUCCESS",{url:location.href,capture_id:record.capture_id,status:result.status,bytes:result.bodyBytes,response:result.responseText});maybeAdvanceKingdom()}else{const failure=describeFailure(result);toast(`Shadow Scrape ✕ ${failure}`,false,8000);console.error("[Shadow Scrape] FAILED",{url:location.href,status:result.status,statusText:result.statusText,response:result.responseText,payloadBytes:result.bodyBytes})}}catch(error){console.error("[Shadow Scrape] exception",error);toast("Shadow Scrape ✕ capture exception",false,7000)}finally{busy=false}}
function scheduleCapture(){setTimeout(capture,CAPTURE_DELAY)}
function hookNavigation(){const p=history.pushState,r=history.replaceState;history.pushState=function(...a){const x=p.apply(this,a);setTimeout(capture,1500);return x};history.replaceState=function(...a){const x=r.apply(this,a);setTimeout(capture,1500);return x};window.addEventListener("popstate",()=>setTimeout(capture,1500));window.addEventListener("hashchange",()=>setTimeout(capture,1500))}
function observePage(){let timer=null;const observer=new MutationObserver(()=>{clearTimeout(timer);timer=setTimeout(()=>{if(document.body&&location.href!==lastUrl)capture()},1200)});if(document.documentElement)observer.observe(document.documentElement,{childList:true,subtree:true})}
function start(){hookNavigation();observePage();scheduleCapture();maybeStartKingdomCycler();console.log("[Shadow Scrape] active",location.hostname)}
start();
})();
